superavatar_0 = {
	entity = "construction_platform_entity"
	construction_entity = "construction_platform_entity"
	# construction_blocks_and_blocked_by = none # BLOCKING TEST
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = no
	prerequisites = { giga_tech_blokkat_project_salvation_psionics_0 }
	
	outliner_trigger = {
		exists = owner
		NOT = { has_megastructure_flag = giga_outliner_hidden_by_@owner }
	}
	
	ai_weight = { factor = 0 }
	build_time = 1800
	resources = {
		category = giga_gigastructures
		cost = {
			influence = 300
			unity = @giga_tera_start_unity_cost
		}
		inline_script = {
			script = megastructures/generic_parts/giga_mega_alloy_cost
			alloys = 10000
		}
		upkeep = { energy = 5 }
	}

	potential = {
		has_technology = giga_tech_blokkat_project_salvation_psionics_0
		giga_has_big_salvationcraft = no
	}

	possible = {
		custom_tooltip = { fail_text = "requires_no_habitable_planets"		NOT = { any_system_planet = { is_colony = yes } } }
		custom_tooltip = { fail_text = "requires_inside_border"				is_inside_border = from }
		custom_tooltip = { fail_text = "requires_no_crisis_system"			NOT = { any_system_planet = { has_planet_flag = crisis_vital_planet } } }
		custom_tooltip = { fail_text = "requires_no_existing_big_salvationcraft"			from = { giga_has_big_salvationcraft = no } }
	}

	placement_rules = {
		planet_possible = {
			custom_tooltip = { fail_text = "requires_no_existing_megastructure"				planet_has_no_megastructure = yes }
			custom_tooltip = { fail_text = "must_build_around_star"							is_star = yes }
			custom_tooltip = { fail_text = "requires_no_anomaly"							has_anomaly = no }
			custom_tooltip = { 
				fail_text = "must_build_around_black_hole"
				giga_is_standard_black_hole = yes
				NOT = {
					giga_is_bitrinary = yes
				}
			}
		}
	}

	on_build_complete = {
		save_event_target_as = giga_system
		if = { limit = { exists = fromfrom.planet } fromfrom.planet = { save_event_target_as = giga_planet } }
		from = {
			country_event = { id = giga_dialog.6001 }	# Notification
			change_variable = { which = giga_current_big_salvationcraft value = 1 }
		}
		fromfrom.planet = {
			if = { limit = { exists = orbital_station } orbital_station = { dismantle = yes } }
			set_carrier_flag = megastructure
			if = { limit = { NOT = { has_planet_flag = shroud_storm_changed } } } set_planet_flag = shroud_storm_changed
			giga_set_has_mega_flag = yes
		}
	}
}

superavatar_4 = {
	entity = "war_system_phase_04_entity"
	construction_entity = "war_system_phase_04_construction_entity"
	portrait = "GFX_megastructure_moon"
	upgrade_from = { superavatar_3 }
	
	outliner_trigger = {
		exists = owner
		NOT = { has_megastructure_flag = giga_outliner_hidden_by_@owner }
	}
	
	build_time = 3600
	resources = {
		category = giga_gigastructures
		cost = {
			unity = @giga_tera_unity_cost
		}
		inline_script = {
			script = megastructures/generic_parts/giga_mega_alloy_cost
			alloys = 100000
		}
		upkeep = {
			energy = 750
			alloys = 500
		}
	}

	possible = {
		from = {
			has_technology = giga_tech_blokkat_project_salvation_psionics_0
			has_technology = giga_tech_blokkat_project_salvation_psionics_1
			has_technology = giga_tech_blokkat_project_salvation_psionics_2
			has_technology = giga_tech_blokkat_project_salvation_psionics_3
			has_technology = giga_tech_blokkat_project_salvation_psionics_4
		}
	}


	on_build_complete = {
		save_event_target_as = giga_system
		if = {
			limit = {
				NOT = {
					any_system_planet = {
						is_star = yes
						NOT = { is_planet_class = pc_giga_empty_space }
					}
				}
			}
			set_star_class = sc_empty_space
		}
		if = { limit = { exists = fromfrom.planet } fromfrom.planet = { save_event_target_as = giga_planet } }
		from = {
			set_country_flag = giga_has_active_hyperweapon
			change_variable = { which = giga_current_big_salvationcraft_finished value = 1 }
			country_event = { id = giga_dialog.6005 }	# Notification
			giga_make_superavatar = yes
		}
		last_created_fleet = { set_location = fromfrom }
		remove_megastructure = fromfrom
	}
}
