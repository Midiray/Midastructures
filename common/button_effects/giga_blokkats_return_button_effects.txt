# disabled -> moves to enabled
blokkat2_x_disabled = {
    potential = {
        exists = from
        from = {
            has_country_flag = giga_menu_page_00
            has_global_flag = giga_blokkat2_disabled
        }
    }
    allow = {
        custom_tooltip = "must_not_have_started_game"
        hidden_trigger = { NOT = { has_global_flag = giga_game_started } }
    }
    effect = {
        custom_tooltip = "giga_blokkat2_disabled"
        hidden_effect = {
            remove_global_flag = giga_blokkat2_disabled
            if = {
                # if no difficulty tag, set a default or else the icons disappear lmao
                limit = {
                    NOR = {
                        has_global_flag = giga_blokkat2_deathmode
                        has_global_flag = giga_blokkat2_endtech
                        has_global_flag = giga_blokkat2_bismuth
                        has_global_flag = giga_blokkat2_normal
                        has_global_flag = giga_blokkat2_easy
                        has_global_flag = giga_blokkat2_very_easy
                    }
                }
                set_blokkat2_difficulty = yes
            }
        }
    }
}
